cmake_minimum_required(VERSION 3.22)
project(folderaudio LANGUAGES CXX)

set(FFMPEG "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.native/ffmpeg/${ANDROID_ABI}")
foreach(component avformat avcodec swresample avutil)
    set(library "${FFMPEG}/lib/lib${component}.so")
    if(NOT EXISTS "${library}")
        message(FATAL_ERROR "Missing FFmpeg libraries. Run Gradle prepareFfmpeg first.")
    endif()
    add_library(${component} SHARED IMPORTED)
    set_target_properties(${component} PROPERTIES IMPORTED_LOCATION "${library}")
endforeach()

add_library(folderaudio SHARED decoder.cpp player.cpp)
target_compile_features(folderaudio PRIVATE cxx_std_17)
target_compile_definitions(folderaudio PRIVATE _FILE_OFFSET_BITS=64)
target_compile_options(folderaudio PRIVATE -Wall -Wextra -fvisibility=hidden)
target_include_directories(folderaudio PRIVATE "${FFMPEG}/include")
target_link_libraries(folderaudio PRIVATE avformat avcodec swresample avutil aaudio android dl)
